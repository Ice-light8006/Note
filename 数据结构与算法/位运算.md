# 将无符号整数转换为二进制字符
```cpp
#include<iostream>
#include<string>

using namespace std;

int main()
{
	string bits;
	unsigned long long n;
	cin >> n;
	if(n==0)
	{
		cout << n << endl;
		return 0;
	}
	while(n>0)
	{
		bits.push_back((n&1)==1?'1':'0');
		n>>=1;
	}
	reverse(bits.begin(),bits.end());
	cout << bits << endl;
	return 0;
}
```

# 图论基础算法模板 (C++)
常用头文件与全局定义
```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll INF = 0x3f3f3f3f3f3f3f3f; // 足够大的数，避免溢出
// 邻接表存图: 节点编号一般为 1~n
// graph[u] 存的是 {v, w}
using pii = pair<int, int>; // {邻接点, 边权}
```
---
## 1. Dijkstra（单源非负权最短路）
- **适用范围**：边权非负
- **复杂度**：`O((V+E)logV)`
- **注意**：不能有负权边
```cpp

vector<ll> dijkstra(int s, const vector<vector<pii>>& adj) {
    int n = adj.size() - 1; // 1-indexed
    vector<ll> dist(n + 1, INF);
    vector<bool> vis(n + 1, false);
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll,int>>> pq;
    dist[s] = 0;
    pq.push({0, s});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto [v, w] : adj[u]) {
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}
```

## 2. Bellman-Ford（单源可负权最短路 & 负环检测）
- **适用范围**：可有负权边，能检测负环
- **复杂度**：`O(VE)`

```cpp

// 返回最短路数组，若存在负环则返回空 vector
vector<ll> bellman_ford(int s, const vector<tuple<int,int,int>>& edges, int n) {
    vector<ll> dist(n + 1, INF);
    dist[s] = 0;
    for (int i = 1; i <= n - 1; ++i) {
        bool updated = false;
        for (auto [u, v, w] : edges) {
            if (dist[u] != INF && dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                updated = true;
            }
        }
        if (!updated) break;
    }
    // 检测负环
    for (auto [u, v, w] : edges) {
        if (dist[u] != INF && dist[v] > dist[u] + w)
            return {}; // 存在负环
    }
    return dist;
}
```

## 3. SPFA（队列优化的 Bellman-Ford）
- **适用范围**：负权图，判断负环
- **最坏复杂度**：`O(VE)`，在实际随机图中通常较快
- **注意**：容易被特殊构造的数据卡掉
```cpp

// 返回最短路数组；若存在负环返回空 vector
vector<ll> spfa(int s, const vector<vector<pii>>& adj) {
    int n = adj.size() - 1;
    vector<ll> dist(n + 1, INF);
    vector<bool> inq(n + 1, false);
    vector<int> cnt(n + 1, 0);      // 记录每个点入队次数
    queue<int> q;
    dist[s] = 0;
    q.push(s);
    inq[s] = true;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        inq[u] = false;
        for (auto [v, w] : adj[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (!inq[v]) {
                    q.push(v);
                    inq[v] = true;
                    if (++cnt[v] >= n) return {}; // 存在负环
                }
            }
        }
    }
    return dist;
}
```

## 4. Floyd（多源最短路）
- **适用范围**：求任意两点间的最短路（可负权，无负环）
- **复杂度**：`O(V³)`
- **注意**：最外层循环必须是中间节点 k
```cpp

// dis 初始化为邻接矩阵，dis[i][i]=0，无边则为 INF
void floyd(vector<vector<ll>>& dis) {
    int n = dis.size() - 1;
    for (int k = 1; k <= n; ++k)
        for (int i = 1; i <= n; ++i)
            for (int j = 1; j <= n; ++j)
                if (dis[i][k] != INF && dis[k][j] != INF)
                    dis[i][j] = min(dis[i][j], dis[i][k] + dis[k][j]);
}
```

## 5. Prim（最小生成树 - 稠密图）
- **适用范围**：无向连通图
- **堆优化复杂度**：`O((V+E)logV)`
- **返回**：最小生成树总权值，不连通返回 -1
```cpp
ll prim(const vector<vector<pii>>& adj) {
    int n = adj.size() - 1;
    vector<bool> vis(n + 1, false);
    vector<ll> min_e(n + 1, INF); // min_e[i] 为点 i 到当前生成树的距离
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<>> pq;
    min_e[1] = 0;
    pq.push({0, 1});
    ll mst_weight = 0;
    int cnt = 0;
    while (!pq.empty()) {
        auto [w, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        mst_weight += w;
        ++cnt;
        for (auto [v, cost] : adj[u]) {
            if (!vis[v] && cost < min_e[v]) {
                min_e[v] = cost;
                pq.push({cost, v});
            }
        }
    }
    return (cnt == n) ? mst_weight : -1;
}
```

## 6. Kruskal（最小生成树 - 稀疏图）
- **适用范围**：无向连通图
- **复杂度**：`O(ElogE)`
- **返回**：最小生成树总权值，不连通返回 -1
```cpp
struct DSU {
    vector<int> parent, rank;
    DSU(int n) : parent(n+1), rank(n+1, 0) {
        for (int i=1; i<=n; ++i) parent[i] = i;
    }
    int find(int x) {
        return parent[x] == x ? x : parent[x] = find(parent[x]);
    }
    void unite(int x, int y) {
        x = find(x), y = find(y);
        if (x == y) return;
        if (rank[x] < rank[y]) parent[x] = y;
        else if (rank[x] > rank[y]) parent[y] = x;
        else parent[y] = x, ++rank[x];
    }
};
struct Edge {
    int u, v, w;
    bool operator<(const Edge& other) const {
        return w < other.w;
    }
};
ll kruskal(int n, vector<Edge> edges) {
    sort(edges.begin(), edges.end());
    DSU dsu(n);
    ll mst_weight = 0;
    int cnt = 0;
    for (auto [u, v, w] : edges) {
        if (dsu.find(u) != dsu.find(v)) {
            dsu.unite(u, v);
            mst_weight += w;
            if (++cnt == n - 1) break;
        }
    }
    return (cnt == n - 1) ? mst_weight : -1;
}
```

## 7. 拓扑排序（Topological Sort）
- **适用范围**：有向无环图（DAG）
- **复杂度**：`O(V+E)`
- **返回**：一个合法的拓扑序列，若存在环则返回空 vector
```cpp

// Kahn 算法（BFS）
vector<int> topological_sort(const vector<vector<int>>& adj) {
    int n = adj.size() - 1;
    vector<int> in_degree(n + 1, 0);
    for (int u = 1; u <= n; ++u)
        for (int v : adj[u])
            ++in_degree[v];
    queue<int> q;
    for (int i = 1; i <= n; ++i)
        if (in_degree[i] == 0) q.push(i);
    vector<int> topo;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        topo.push_back(u);
        for (int v : adj[u]) {
            if (--in_degree[v] == 0)
                q.push(v);
        }
    }
    // 如果 topo 的大小 < n，说明图中有环
    if (topo.size() != n) return {};
    return topo;
}
```

### 补充说明
- 以上代码均采用 **1-indexed** 节点编号。
- **INF** 通常设为 `0x3f3f3f3f` 或 `1e18`，可根据实际最大值调整。
- 涉及负权边时，请使用`Bellman-Ford`或 `SPFA`，`Dijkstra`不适用。
- `SPFA `虽然平均快，但在竞赛中有可能被卡，建议用 Dijkstra + 势能（Johnson）处理含负权图的最短路。

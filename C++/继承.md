# 继承权限
继承权限有`public`,`protected`,`private`
这个表格介绍了这三个继承的作用
第一列为某一个成员在基类中的权限，第一行为派生类的继承方式
第二行到第四行的第二列到第四列为派生类的访问权限

| 成员在基类中的权限   | `public`继承 | `protected`继承 | `private`继承 |
| ----------- | ---------- | ------------- | ----------- |
| `public`    | `public`   | `protected`   | 派生类不可直接访问   |
| `protected` | `public`   | `protected`   | 派生类不可直接访问   |
| `private`   | 派生类不可直接访问  | 派生类不可直接访问     | 派生类不可直接访问   |
# 不能被继承的成员
构造函数、析构函数、静态函数

这种继承结构我们常称之为菱形继承
其中人类有成员变量`name`，医生有特有的成员变量`title`（表示职称），军人有特有的成员变量`rank`（表示军衔）
用下面这个代码表示
```cpp
#include <iostream>
#include <string>

using namespace std;

class Human
{
private:
    string name;
public:
    Human(){name = "空白";cout << "Human的默认构造函数被调用" << endl;}
    Human(string name):name(name)
    {
        cout << "Human的带参构造函数被调用" << endl;
    }
    string GetName(){return name;}
    ~Human(){cout << "人类的析构函数被调用" << endl;}
};

class Doctor:public Human
{
private:
    string title;
public:
    Doctor()
    {
        title = "空白";
        cout << "医生的默认构造函数调用" << endl;
    }
    Doctor(string name,string title):Human(name),title(title){cout << "医生的带参构造函数被调用" << endl;}
    string GetTitle(){return title;}
    ~Doctor(){cout << "医生的析构函数被调用" << endl;}
};

class Armyman:public Human
{
private:
    string rank;
public:
    Armyman(){rank = "空白";cout << "军人的默认构造函数被调用" << endl;}
    Armyman(string name,string rank):Human(name),rank(rank){cout << "军人的带参构造函数被调用" << endl;}
    string GetRank(){return rank;}
    ~Armyman(){cout << "军人的析构函数被调用" << endl;} 
};

class MilitaryDoctor:public Doctor,public Armyman
{
private:
public:
    MilitaryDoctor(){cout << "军医默认构造函数被调用" << endl;}
    MilitaryDoctor(string name1,string title,string name2,string rank)
    :Doctor(name1,title),Armyman(name2,rank)
    {
        cout << "军医带参构造函数被调用" << endl;
    }
    void show()
    {
        cout << Doctor::GetName() << ' ' << GetTitle() << ',' << Armyman::GetName() << ' ' << GetRank() << endl;
    }
    ~MilitaryDoctor(){cout << "军医的析构函数被调用" << endl;}

};

int main()
{
    cout << "========== 开始 ==========" << endl;
    MilitaryDoctor md("张三","主治医师","李四","上校");
    md.show();
    cout << "========== 结束 ==========" << endl;
    return 0;
}
```

```Output
军医带参构造函数被调用
```

但是这里有一个很关键的问题，`Doctor` 继承来一个`name`，`Armyman`继承来一个`name`
那军医的名字到底是医生的名字还是军人的名字？
这就产生了二义性
可以看到我们的代码中使用了`name1`和`name2`两个名字分别给医生和军人初始化
在访问`GetName()`函数时用了作用域分辨符`Doctor::`和`Armyman::`来区分作用域，进而区分医生继承来的`Human`类的`name`和军人继承来的`Human`类的`name`
那么产生这个问题的本质是什么呢？实际上是因为在 `MilitaryDoctor` 对象内部，`Doctor` 基类子对象和 `Armyman` 基类子对象分别拥有一个独立的 `Human` 基类子对象。示意图如下

可以发现`Doctor`的`Human`基类子对象的地址和`Armyman`的`Human`的基类子对象的地址不同。
我们只要让`Doctor`对象和 `Armyman`对象共享一个`Human`基类子对象就可以了。为了解决这个问题，C++引入了**虚继承**和**虚基类**的概念
虚基类的概念是伴随着虚继承的定义而产生的。虚继承的语法定义如下
```cpp
class 派生类名:virtual 继承方式 基类名
```
这里`virtual`是一个关键字。
把刚刚的代码修改为下面这样

可以发现`Human`只构造了一次。`Doctor`与`Armyman`共享这一个`Human`基类子对象
但是注意一个问题，我们执行了下面这个代码来构造一个军医对象
```cpp
MilitaryDoctor md("张三","主治医师","李四","上校");
```
`MilitaryDoctor`构造函数的初始化列表执行了
```cpp
Doctor("张三","主治医师");
Armyman("李四","上校");
```
而`Doctor`构造函数的初始化列表又调用了
```cpp
Human("张三");
```
`Armyman`的构造函数的初始化列表调用了
```cpp
Human("李四");
```
那么虚基类的`name`的值应该为张三或者李四才对。为什么是空白呢？而这个空白来自于`Human`的默认构造函数。也就是说
```cpp
Human("张三");
Human("李四");
```
这两行代码压根没有被执行。而是执行了默认构造函数：
```cpp
Human();
```
实际上，在非最派生类的初始化列表中调用的虚基类的构造函数会全部被忽略。也就是说，这里
```cpp
Doctor(string name,string title):Human(name),title(title){cout << "医生的带参构造函数被调用" << endl;}
```
等价于
```cpp
Doctor(string name,string title):title(title){cout << "医生的带参构造函数被调用" << endl;}
```
而虚基类子对象的初始化由最派生类的构造函数负责，无论该虚基类是最派生类的直接基类还是间接基类。
如果虚基类没有写任何构造函数，则会自动生成默认构造函数。在有默认构造函数的情况下（无论是系统生成的还是自己写的）此时如果你不显式在最派生类的构造函数的初始化列表中显式调用虚基类的默认构造函数，编译器会自动补上虚基类的构造函数
如果虚基类有其他类型的构造函数但是没有默认构造函数，此时必须要在最派生类显式调用虚基类的构造函数。否则编译器会报错。
所以，刚刚实际上是在最派生类调用了虚基类的默认构造函数构造的，进而得到了虚基类的`name`是空白。
所以，最终解决方案如下：
```cpp
#include <iostream>
#include <string>

using namespace std;

class Human
{
private:
    string name;
public:
    Human(){name = "空白";cout << this << " Human的默认构造函数被调用" << endl;}
    Human(string name):name(name)
    {
        cout << this << " Human的带参构造函数被调用" << endl;
    }
    string GetName(){return name;}
    ~Human(){cout << "人类的析构函数被调用" << endl;}
};

class Doctor:virtual public Human
{
private:
    string title;
public:
    Doctor()
    {
        title = "空白";
        cout << "医生的默认构造函数调用" << endl;
    }
    Doctor(string name,string title):Human(name),title(title){cout << "医生的带参构造函数被调用" << endl;}
    string GetTitle(){return title;}
    ~Doctor(){cout << "医生的析构函数被调用" << endl;}
};

class Armyman:virtual public Human
{
private:
    string rank;
public:
    Armyman(){rank = "空白";cout << "军人的默认构造函数被调用" << endl;}
    Armyman(string name,string rank):Human(name),rank(rank){cout << "军人的带参构造函数被调用" << endl;}
    string GetRank(){return rank;}
    ~Armyman(){cout << "军人的析构函数被调用" << endl;} 
};

class MilitaryDoctor:public Doctor,public Armyman
{
private:
public:
    MilitaryDoctor(){cout << "军医默认构造函数被调用" << endl;}
    MilitaryDoctor(string name,string title,string rank)//修改为只传一个名字
    :Doctor(name,title),Armyman(name,rank),Human(name)//显式调用Human的带参构造函数，用name构造Human
    {
        cout << "军医带参构造函数被调用" << endl;
    }
    void show()
    {
        cout << GetName() << ' ' << GetTitle() << ' ' << GetRank() << endl;
    }
    ~MilitaryDoctor(){cout << "军医的析构函数被调用" << endl;}

};

int main()
{
    cout << "========== 开始 ==========" << endl;
    MilitaryDoctor md("张三","主治医师","上校");
    md.show();
    cout << "========== 结束 ==========" << endl;
    return 0;
}
```

下面这个是我们需要烧录进ESP8266模块的代码
这段代码的作用是在8080端口建立一个TCP服务器，并在 Wi-Fi 和硬件串口之间做数据的双向搬运。
```cpp
#include <ESP8266WiFi.h>

const char* ssid = "Xiaomi 15 Pro";
const char* password = "你的WiFi密码";
const int port = 8080; // 电脑端 Qt 客户端需要连接的端口

WiFiServer server(port);
WiFiClient client;

void setup() {
  // 这里的波特率必须与你 STM32 串口配置的波特率完全一致
  Serial.begin(115200); 
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  server.begin();
}

void loop() {
  // 检查是否有电脑端客户端（比如你的 Qt 软件）连入
  if (!client.connected()) {
    client = server.available();
  } else {
    // 方向 1: Wi-Fi 收到电脑发来的数据 -> 转发给 STM32 (串口)
    if (client.available()) {
      size_t len = client.available();
      uint8_t sbuf[len];
      client.readBytes(sbuf, len);
      Serial.write(sbuf, len);
    }
    
    // 方向 2: 串口收到 STM32 发来的数据 -> 转发给电脑 (Wi-Fi)
    if (Serial.available()) {
      size_t len = Serial.available();
      uint8_t sbuf[len];
      Serial.readBytes(sbuf, len);
      client.write(sbuf, len);
    }
  }
}
```
这里`setup`函数只在开机运行一次（每次按下复位按键之后也会运行一次）
`loop`函数会在开机之后直接一直重复调用
整个程序相当于
```cpp
int main()
{
	setup();
	while(1)
	{
		loop();
	}
}
```

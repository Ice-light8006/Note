下面这个是我们需要烧录进ESP8266模块的代码
这段代码的作用是在8080端口建立一个TCP服务器，并在 Wi-Fi 和硬件串口之间做数据的双向搬运。
```cpp
#include <ESP8266WiFi.h>

const char* ssid = "Xiaomi 15 Pro";
const char* password = "你的WiFi密码";
const int port = 8080; // 电脑端 Qt 客户端需要连接的端口

WiFiServer server(port);
WiFiClient client;

void setup() {
  // 这里的波特率必须与你 STM32 串口配置的波特率完全一致
  Serial.begin(115200); 
  
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }
  server.begin();
}

void loop() {
  // 检查是否有电脑端客户端（比如你的 Qt 软件）连入
  if (!client.connected()) {
    client = server.available();
  } else {
    // 方向 1: Wi-Fi 收到电脑发来的数据 -> 转发给 STM32 (串口)
    if (client.available()) {
      size_t len = client.available();
      uint8_t sbuf[len];
      client.readBytes(sbuf, len);
      Serial.write(sbuf, len);
    }
    
    // 方向 2: 串口收到 STM32 发来的数据 -> 转发给电脑 (Wi-Fi)
    if (Serial.available()) {
      size_t len = Serial.available();
      uint8_t sbuf[len];
      Serial.readBytes(sbuf, len);
      client.write(sbuf, len);
    }
  }
}
```
这里`setup`函数只在开机运行一次（每次按下复位按键之后也会运行一次）
`loop`函数会在开机之后直接一直重复调用
整个程序相当于
```cpp
int main()
{
	setup();
	while(1)
	{
		loop();
	}
}
```


新的代码
```cpp
#include <ESP8266WiFi.h>

// Wi-Fi
const char* ssid = "407Lab";
const char* password = "12345678";

// TCP server port
const uint16_t port = 8080;

// STM32 USART3 and ESP8266 UART must both use 115200, 8N1.
const uint32_t UART_BAUD = 115200;
const size_t BUFFER_SIZE = 128;

WiFiServer server(port);
WiFiClient client;

uint8_t buffer[BUFFER_SIZE];

void acceptNewClient()
{
    // ESP8266 core's server.available() acts as accepting a waiting client.
    // Always check it, so a new Qt connection can replace an old one.
    WiFiClient newClient = server.available();

    if (!newClient) {
        return;
    }

    // This system only supports one PC client.
    // If the PC reconnects, discard the old connection and use the new one.
    if (client && client.connected()) {
        client.stop();
    }

    client = newClient;
    client.setNoDelay(true);
}

void forwardTcpToStm32()
{
    while (client.available() > 0) {
        size_t length = 0;

        // Read TCP bytes without variable-length stack arrays or blocking waits.
        while (client.available() > 0 && length < BUFFER_SIZE) {
            int value = client.read();
            if (value < 0) {
                break;
            }

            buffer[length++] = static_cast<uint8_t>(value);
        }

        if (length == 0) {
            return;
        }

        // Qt -> ESP8266 -> STM32
        Serial.write(buffer, length);
        yield();
    }
}

void forwardStm32ToTcp()
{
    while (Serial.available() > 0) {
        size_t length = 0;

        // Read bytes currently received from STM32.
        while (Serial.available() > 0 && length < BUFFER_SIZE) {
            int value = Serial.read();
            if (value < 0) {
                break;
            }

            buffer[length++] = static_cast<uint8_t>(value);
        }

        if (length == 0 || !client.connected()) {
            return;
        }

        // STM32 -> ESP8266 -> Qt
        size_t sent = 0;
        while (sent < length && client.connected()) {
            size_t current = client.write(buffer + sent, length - sent);

            if (current == 0) {
                return;
            }

            sent += current;
            yield();
        }
    }
}

void setup()
{
    Serial.begin(115200);

    WiFi.begin(ssid, password);

    Serial.print("正在连接 Wi-Fi");

    while (WiFi.status() != WL_CONNECTED)
    {
        delay(500);
        Serial.print(".");
    }

    server.begin();
}
void loop()
{
    // Always accept a newly connected Qt client.
    acceptNewClient();

    if (!client || !client.connected()) {
        if (client) {
            client.stop();
        }

        yield();
        return;
    }

    forwardTcpToStm32();
    forwardStm32ToTcp();

    yield();
}
```
